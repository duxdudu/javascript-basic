🖱️ Events

Events are things that happen in the browser — a click, a key press, a form submit. JavaScript lets you respond to them.

1. addEventListener — the right way to handle events

const btn = document.getElementById("myBtn");

// element.addEventListener("eventName", function)
btn.addEventListener("click", function() {
  console.log("Button was clicked!");
});

// Arrow function version
btn.addEventListener("click", () => {
  console.log("Clicked!");
});

2. Common event types

// Mouse events
element.addEventListener("click",      () => { /* single click   */ });
element.addEventListener("dblclick",   () => { /* double click   */ });
element.addEventListener("mouseover",  () => { /* hover over     */ });
element.addEventListener("mouseout",   () => { /* hover away     */ });

// Keyboard events
input.addEventListener("keydown",  (e) => { console.log(e.key); });
input.addEventListener("keyup",    (e) => { console.log(e.key); });

// Input / Form events
input.addEventListener("input",    (e) => { console.log(e.target.value); });
input.addEventListener("change",   (e) => { /* value changed    */ });
form.addEventListener("submit",    (e) => { e.preventDefault(); /* stop page reload */ });

3. The event object

const btn = document.getElementById("myBtn");

btn.addEventListener("click", function(event) {
  // 'event' contains info about what happened
  console.log(event.type);   // "click"
  console.log(event.target); // the element that was clicked
});

// Keyboard event — which key was pressed?
document.addEventListener("keydown", function(event) {
  console.log(event.key);    // "Enter", "a", "ArrowUp", etc.

  if (event.key === "Enter") {
    console.log("Enter was pressed!");
  }
});

4. Preventing default behaviour

const form = document.querySelector("form");

form.addEventListener("submit", function(event) {
  event.preventDefault(); // stops the page from reloading

  // Now handle the form data yourself
  const name = document.querySelector("#name").value;
  console.log("Submitted:", name);
});

// Also useful on links
const link = document.querySelector("a");
link.addEventListener("click", function(event) {
  event.preventDefault(); // stops navigation
  console.log("Link clicked but not followed");
});

5. Removing an event listener

function handleClick() {
  console.log("Clicked!");
}

btn.addEventListener("click", handleClick);

// Later — remove it
btn.removeEventListener("click", handleClick);
// Note: must pass the same named function, not an anonymous one
