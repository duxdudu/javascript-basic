📋 Arrays

An array is an ordered list of values. It's one of the most used data structures in JavaScript.

1. Creating an array

const fruits  = ["apple", "banana", "cherry"];
const numbers = [10, 20, 30, 40];
const mixed   = ["Alice", 25, true]; // can mix types

console.log(fruits);    // ["apple", "banana", "cherry"]
console.log(fruits[0]); // "apple"   — index starts at 0
console.log(fruits[2]); // "cherry"
console.log(fruits.length); // 3

2. Adding and removing items

const items = ["a", "b", "c"];

items.push("d");      // add to END   → ["a","b","c","d"]
items.unshift("z");   // add to START → ["z","a","b","c","d"]

items.pop();          // remove from END   → ["z","a","b","c"]
items.shift();        // remove from START → ["a","b","c"]

console.log(items); // ["a","b","c"]

3. Looping through an array

const colors = ["red", "green", "blue"];

// for loop
for (let i = 0; i < colors.length; i++) {
  console.log(colors[i]);
}

// for...of (cleaner)
for (const color of colors) {
  console.log(color);
}

// forEach
colors.forEach(function(color) {
  console.log(color);
});

4. map — transform every item

const numbers = [1, 2, 3, 4, 5];

// Double every number
const doubled = numbers.map(function(n) {
  return n * 2;
});

console.log(doubled); // [2, 4, 6, 8, 10]
// Original array is NOT changed
console.log(numbers); // [1, 2, 3, 4, 5]

5. filter — keep only matching items

const numbers = [1, 2, 3, 4, 5, 6];

const evens = numbers.filter(function(n) {
  return n % 2 === 0; // keep if remainder is 0
});

console.log(evens); // [2, 4, 6]

// Real-world example — filter completed tasks
const tasks = [
  { text: "Buy milk",  completed: true  },
  { text: "Read book", completed: false },
  { text: "Exercise",  completed: true  },
];

const done = tasks.filter(task => task.completed === true);
console.log(done.length); // 2

6. find — get the first match

const users = [
  { id: 1, name: "Alice" },
  { id: 2, name: "Bob"   },
  { id: 3, name: "Carol" },
];

const user = users.find(function(u) {
  return u.id === 2;
});

console.log(user); // { id: 2, name: "Bob" }

7. includes — check if a value exists

const fruits = ["apple", "banana", "cherry"];

console.log(fruits.includes("banana")); // true
console.log(fruits.includes("grape"));  // false

8. join and split

const words = ["Hello", "World", "JS"];

// Array → String
const sentence = words.join(" ");
console.log(sentence); // "Hello World JS"

// String → Array
const parts = "a,b,c,d".split(",");
console.log(parts); // ["a","b","c","d"]
