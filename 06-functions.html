⚙️ Functions

A function is a reusable block of code. You define it once and call it as many times as you need.

1. Declaring and calling a function

// Define the function
function sayHello() {
  console.log("Hello!");
}

// Call (run) the function
sayHello(); // Hello!
sayHello(); // Hello! — can call it many times

2. Parameters — passing data in

function greet(name) {
  console.log("Hello, " + name + "!");
}

greet("Alice"); // Hello, Alice!
greet("Bob");   // Hello, Bob!

// Multiple parameters
function add(a, b) {
  console.log(a + b);
}
add(3, 7); // 10

3. Return values — getting data out

function multiply(a, b) {
  return a * b; // sends the result back
}

const result = multiply(4, 5);
console.log(result); // 20

// Without return, a function gives back undefined
function noReturn() {
  const x = 10;
}
console.log(noReturn()); // undefined

4. Default parameters

function greet(name = "stranger") {
  return "Hello, " + name + "!";
}

console.log(greet("Alice")); // Hello, Alice!
console.log(greet());        // Hello, stranger!

5. Function Expression — storing a function in a variable

const square = function(n) {
  return n * n;
};

console.log(square(4)); // 16
console.log(square(9)); // 81

6. Arrow Functions — shorter syntax

// Regular function
function double(n) {
  return n * 2;
}

// Arrow function — same thing, shorter
const double = (n) => n * 2;

// Even shorter when there's one parameter
const double = n => n * 2;

console.log(double(5)); // 10

// Multi-line arrow function needs curly braces + return
const greet = (name) => {
  const message = "Hello, " + name;
  return message;
};

7. Functions calling other functions

function celsiusToFahrenheit(c) {
  return (c * 9/5) + 32;
}

function showTemperature(celsius) {
  const fahrenheit = celsiusToFahrenheit(celsius); // call another function
  console.log(celsius + "°C = " + fahrenheit + "°F");
}

showTemperature(0);   // 0°C = 32°F
showTemperature(100); // 100°C = 212°F
