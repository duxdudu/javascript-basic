🗂️ Objects

An object groups related data together using key-value pairs. Think of it like a form with labelled fields.

1. Creating an object

const person = {
  name: "Alice",      // key: value
  age: 25,
  city: "New York",
  isStudent: false
};

console.log(person);        // the whole object
console.log(person.name);   // "Alice"  — dot notation
console.log(person["age"]); // 25       — bracket notation

2. Adding and updating properties

const car = {
  brand: "Toyota",
  year: 2020
};

// Update existing property
car.year = 2024;

// Add a new property
car.color = "blue";

console.log(car);
// { brand: "Toyota", year: 2024, color: "blue" }

3. Methods — functions inside objects

const person = {
  name: "Alice",
  age: 25,

  // A method is a function stored as a property
  greet: function() {
    return "Hi, I'm " + this.name + "!";
  },

  isAdult: function() {
    return this.age >= 18;
  }
};

console.log(person.greet());   // "Hi, I'm Alice!"
console.log(person.isAdult()); // true

4. Looping through an object

const scores = {
  math: 90,
  english: 85,
  science: 92
};

// for...in loops over the keys
for (const subject in scores) {
  console.log(subject + ": " + scores[subject]);
}
// math: 90
// english: 85
// science: 92

// Get all keys as an array
console.log(Object.keys(scores));   // ["math","english","science"]
console.log(Object.values(scores)); // [90, 85, 92]

5. Array of objects — very common pattern

const students = [
  { name: "Alice", grade: "A" },
  { name: "Bob",   grade: "B" },
  { name: "Carol", grade: "A" },
];

// Loop through and print each student
students.forEach(function(student) {
  console.log(student.name + " got a " + student.grade);
});

// Filter only A students
const topStudents = students.filter(s => s.grade === "A");
console.log(topStudents.length); // 2

6. Spread operator — copy or merge objects

const original = { name: "Alice", age: 25 };

// Copy an object (so you don't change the original)
const copy = { ...original };
copy.age = 30;

console.log(original.age); // 25 — unchanged
console.log(copy.age);     // 30

// Merge two objects
const extra = { city: "Paris" };
const merged = { ...original, ...extra };
console.log(merged); // { name:"Alice", age:25, city:"Paris" }
