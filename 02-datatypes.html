🔢 Data Types

Every value in JavaScript has a type. Knowing the types helps you understand what you can do with a value.

1. String — text

const firstName = "Alice";          // double quotes
const lastName  = 'Smith';          // single quotes — same thing
const greeting  = `Hello, ${firstName}!`; // template literal (backtick)

console.log(greeting);   // Hello, Alice!
console.log(firstName.length); // 5  — number of characters
console.log(firstName.toUpperCase()); // ALICE

2. Number — integers and decimals

const age    = 25;       // whole number
const price  = 9.99;     // decimal
const result = 10 / 3;   // 3.3333...

console.log(typeof age); // "number"

// Useful number methods
console.log(Math.round(result));  // 3
console.log(Math.floor(9.9));     // 9
console.log(Math.ceil(9.1));      // 10
console.log(Math.max(5, 10, 3));  // 10

3. Boolean — true or false

const isLoggedIn = true;
const hasError   = false;

console.log(isLoggedIn); // true
console.log(typeof isLoggedIn); // "boolean"

// Booleans come from comparisons
const isAdult = age >= 18;  // true if age is 18 or more
console.log(isAdult); // true (age is 25)

4. null — intentionally empty

let selectedUser = null; // we haven't chosen a user yet

// Later we assign a real value
selectedUser = "Alice";
console.log(selectedUser); // Alice

5. undefined — not yet assigned

let score;               // declared but no value given
console.log(score);      // undefined

function greet(name) {
  console.log(name);     // undefined if called without argument
}
greet(); // undefined

6. Array — a list of values

const fruits = ["apple", "banana", "cherry"];

console.log(fruits[0]);      // "apple"  (index starts at 0)
console.log(fruits.length);  // 3
console.log(fruits[fruits.length - 1]); // "cherry" — last item

7. Object — grouped data

const person = {
  name: "Alice",
  age: 25,
  isStudent: true
};

console.log(person.name);      // "Alice"
console.log(person["age"]);    // 25  — bracket notation also works

8. typeof — check a type

console.log(typeof "hello");    // "string"
console.log(typeof 42);         // "number"
console.log(typeof true);       // "boolean"
console.log(typeof undefined);  // "undefined"
console.log(typeof null);       // "object"  ← known JS quirk
console.log(typeof []);         // "object"
console.log(typeof {});         // "object"
